function createGuest() { const sk = skins[Math.floor(Math.random() * skins.length)]; guests.push({ x: W + 45, y: H * .66 + Math.random() * H * .12, size: Math.min(72, H * .09), speed: .7 + Math.random() * .5 + phase * .04, skin: sk, complaint: complaints[Math.floor(Math.random() * complaints.length)], bubble: 100 }); } function bubble(x, y, t) { ctx.save(); ctx.font = "bold 10px Arial"; const w = Math.min(W * .6, ctx.measureText(t).width + 18); const px = Math.max(5, Math.min(W - w - 5, x - w / 2)); rounded(px, y - 31, w, 27, 8, "#fff8e9"); ctx.fillStyle = "#38251e"; ctx.textAlign = "center"; ctx.fillText(t, px + w / 2, y - 13); ctx.restore(); } function drawGuests() { for (const g of guests) { drawPerson( g.x, g.y, g.size, g.skin[0], g.skin[1], g.skin[2], true ); if (g.bubble > 0) { bubble(g.x, g.y - g.size * .65, g.complaint); } } } function shoot() { if (!running) return; if (ammo <= 0) { showMessage("SEM FLECHAS!"); return; } ammo--; const ox = W * .12; const oy = H * .78; const ang = Math.atan2( pointer.y - oy, pointer.x - ox ); arrows.push({ x: ox, y: oy, vx: Math.cos(ang) * 15, vy: Math.sin(ang) * 15, life: 100 }); hud(); } function drawBow() { ctx.save(); ctx.translate(W * .12, H * .78); ctx.rotate( Math.atan2( pointer.y - H * .78, pointer.x - W * .12 ) ); ctx.strokeStyle = "#e8b66f"; ctx.lineWidth = 7; ctx.beginPath(); ctx.arc(0, 0, 32, -1.15, 1.15); ctx.stroke(); ctx.restore(); } function drawArrows() { for (const a of arrows) { ctx.save(); ctx.translate(a.x, a.y); ctx.rotate(Math.atan2(a.vy, a.vx)); line(-17, 0, 19, 0, "#f1d29a", 4); ctx.fillStyle = "#ffd17d"; ctx.beginPath(); ctx.moveTo(24, 0); ctx.lineTo(13, -5); ctx.lineTo(13, 5); ctx.fill(); ctx.restore(); } } function burst(x, y) { for (let i = 0; i < 12; i++) { const a = Math.random() * Math.PI * 2; const v = 1 + Math.random() * 3; particles.push({ x, y, vx: Math.cos(a) * v, vy: Math.sin(a) * v, life: 24, color: i % 2 ? "#ff687b" : "#f4cb81" }); } } function update(dt) { if (!running) return; time += dt; spawnTimer += dt; // Movimento dos hóspedes for (let i = guests.length - 1; i >= 0; i--) { const g = guests[i]; g.x -= g.speed * dt; g.bubble -= dt; if (g.bubble <= 0) { g.bubble = 150 + Math.random() * 80; } // Cliente saiu do hotel if (g.x < -g.size) { guests.splice(i, 1); lives--; showMessage("UM CLIENTE SAIU INSATISFEITO!"); hud(); if (lives <= 0) { gameOver(); return; } } } // Movimento e colisão das flechas for (let i = arrows.length - 1; i >= 0; i--) { const a = arrows[i]; a.x += a.vx * dt; a.y += a.vy * dt; a.life -= dt; let hit = false; for (let j = guests.length - 1; j >= 0; j--) { const g = guests[j]; const distance = Math.hypot( a.x - g.x, a.y - (g.y - g.size * .35) ); if (distance < g.size * .48) { guests.splice(j, 1); score += 100; kills++; burst(a.x, a.y); hit = true; if (score > best) { best = score; try { localStorage.setItem( "hotelHunterRecord", String(best) ); } catch(e) {} } if (kills >= needed) { nextPhase(); } showMessage("RECLAMAÇÃO RESOLVIDA! +100"); hud(); break; } } if ( hit || a.life <= 0 || a.x > W + 40 || a.y > H + 40 || a.y < -40 ) { arrows.splice(i, 1); } } // Partículas for (const p of particles) { p.x += p.vx * dt; p.y += p.vy * dt; p.vy += .08 * dt; p.life -= dt; } particles = particles.filter(p => p.life > 0); // Entrada de novos hóspedes if (spawnTimer > 115 && guests.length < 8) { createGuest(); spawnTimer = 0; } } function draw() { ctx.clearRect(0, 0, W, H); drawRoom(); drawGuests(); drawArrows(); // Efeitos visuais for (const p of particles) { ctx.globalAlpha = Math.max(0, p.life / 24); ellipse(p.x, p.y, 3, 3, p.color); } ctx.globalAlpha = 1; drawBow(); // Mira if (running) { ctx.strokeStyle = "#ff607b"; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(pointer.x, pointer.y, 13, 0, Math.PI * 2); ctx.moveTo(pointer.x - 19, pointer.y); ctx.lineTo(pointer.x + 19, pointer.y); ctx.moveTo(pointer.x, pointer.y - 19); ctx.lineTo(pointer.x, pointer.y + 19); ctx.stroke(); } } function hud() { $("level").textContent = phase; $("points").textContent = score; $("record").textContent = best; $("ammo").textContent = ammo; $("lives").textContent = "♥".repeat( Math.max(0, lives) ); $("phase").childNodes[0].textContent = `FASE ${phase} · ${kills}/${needed}`; $("fill").style.width = Math.min(100, kills / needed * 100) + "%"; } function showMessage(t) { $("message").textContent = t; $("message").style.opacity = 1; clearTimeout(noticeTimer); noticeTimer = setTimeout(() => { $("message").style.opacity = 0; }, 1800); } function nextPhase() { phase++; kills = 0; needed = 7 + phase * 2; ammo += 25; lives = Math.min(5, lives + 1); hud(); showMessage("NOVA FASE! O HOTEL PRECISA DE TI!"); } function startGame() { phase = 1; score = 0; lives = 5; ammo = 65; kills = 0; needed = 7; guests = []; arrows = []; particles = []; spawnTimer = 0; time = 0; running = true; paused = false; $("menu").style.display = "none"; createGuest(); createGuest(); hud(); showMessage("BEM-VINDO AO HOTEL!"); } function showMenu(title, body, button, action) { $("menu").style.display = "flex"; $("panel").innerHTML = `

${title}

${body}

`; $("menuAction").onclick = action; } function pauseGame() { if (!running) return; running = false; paused = true; showMenu( 'JOGO PAUSADO', 'A receção está à tua espera!', 'CONTINUAR', () => { paused = false; running = true; $("menu").style.display = "none"; } ); } function gameOver() { running = false; paused = false; showMenu( 'FIM DE JOGO', `O hotel ficou sem vidas!

PONTOS: ${score}
FASE: ${phase}
RECORDE: ${best}`, 'JOGAR NOVAMENTE', startGame ); } // Ciclo principal function loop(now) { const dt = Math.min( (now - last) / 16.667 || 1, 2 ); last = now; update(dt); draw(); requestAnimationFrame(loop); } // Controlos do rato e do toque function setPointer(e) { const r = canvas.getBoundingClientRect(); pointer.x = e.clientX - r.left; pointer.y = e.clientY - r.top; } canvas.addEventListener("pointermove", setPointer); canvas.addEventListener("pointerdown", e => { setPointer(e); if (running) { shoot(); } }); // Botão de disparo $("fire").addEventListener("pointerdown", e => { e.preventDefault(); e.stopPropagation(); shoot(); }); // Botões principais $("start").onclick = startGame; $("exit").onclick = pauseGame; // Teclado window.addEventListener("keydown", e => { if (e.code === "Space") { e.preventDefault(); shoot(); } if (e.key === "Escape") { pauseGame(); } }); // Inicialização hud(); requestAnimationFrame(loop);
